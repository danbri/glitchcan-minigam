Shared Scene

Multiple creatures in one 3D view with mode="shared"

Usage

<!-- Import the component -->
<script type="module" src="yeti-creature.js"></script>

<!-- Wrap in yeti-scene (loads def once) -->
<yeti-scene>
  <yeti-dog></yeti-dog>
  <yeti-cat></yeti-cat>
  <yeti-elephant color="pink"></yeti-elephant>
</yeti-scene>

<!-- Or standalone (each loads def) -->
<yeti-dog color="white"></yeti-dog>

<!-- Shared 3D scene -->
<yeti-scene mode="shared" width="800" height="400">
  <yeti-dog pos="-3,0,0"></yeti-dog>
  <yeti-cat pos="0,0,0"></yeti-cat>
  <yeti-elephant pos="4,0,0" color="pink"></yeti-elephant>
</yeti-scene>

Attributes

// Display
width        Canvas width (default: 400)
height       Canvas height (default: 300)
spin         Auto-rotate speed (default: 0)

// Colors (named, hex, or r,g,b)
color        Main body color
             Named: white, black, brown, golden, cream, orange, pink, grey
             Hex: #8B4513
             RGB: 0.5,0.3,0.2

// Body proportions (x,y,z)
body-radii   Main body size
head-radii   Head size
ear-radii    Ear size
tail-len     Tail length
tail-r       Tail radius
smooth       Blend smoothness (0.1-0.3)

Advanced: Full Control

<!-- Use base component with all params -->
<yeti-creature
  color="#8B4513"
  body-radii="0.6,0.5,1.0"
  head-radii="0.3,0.28,0.35"
  ear-radii="0.1,0.2,0.05"
  tail-len="0.6"
  smooth="0.2"
></yeti-creature>
